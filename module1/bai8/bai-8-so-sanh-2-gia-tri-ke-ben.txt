import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) {

        List<Double> listDouble = Arrays.asList(9.7, 11.5, 13.3, 5.7, 14.7, 8.8);
        List<Double> topList = new ArrayList<>();
        List<Double> bottomList = new ArrayList<>();

        int length = listDouble.size();
        for (int i=1; i <= length - 2; i++) {
            double left = listDouble.get(i-1);
            double mid = listDouble.get(i);
            double right = listDouble.get(i+1);

            if (mid < left && mid < right) {
                bottomList.add(mid);
                System.out.println("bottomList: " + mid);
            } else if (mid > left && mid > right) {
                topList.add(mid);
                System.out.println("topList: " + mid);
            }
        }

        double sumTop = topList.stream().mapToDouble(Double::doubleValue).sum();
        double sumBottom = bottomList.stream().mapToDouble(Double::doubleValue).sum();
        System.out.println("sumTop: " + sumTop);
        System.out.println("sumBottom: " + sumBottom);

        System.out.println();

    }


}
